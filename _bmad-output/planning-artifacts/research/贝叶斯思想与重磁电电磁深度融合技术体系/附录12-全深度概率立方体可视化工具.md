# 附录12 全深度概率立方体可视化工具

## 概述

全深度概率立方体可视化工具是贝叶斯思想与重磁电电磁深度融合技术体系的核心输出终端，负责将高维的岩性概率分布、不确定性量化结果转化为直观的三维可视化产品，支撑地质解释、勘查决策与成果汇报。

本工具采用模块化设计，支持多种可视化模式、交互操作与输出格式，满足不同用户角色的需求。

> **空间数据契约**：每个概率立方体必须同时记录水平CRS、垂向CRS/基准、正高/椭球高类型、轴序、各轴单位、z轴正方向、地球水准面模型（如适用）、网格到世界坐标变换和转换误差。单一 `coordinate_system = "EPSG:..."` 字段不足以保证三维数据可复现。

> **WP4显示与导出门。** 资源相关视图统一命名为“物探约束靶体潜力情景”，分别显示`volume_m3`、`tonnage_t`、`in_situ_metal_t`和`recovered_metal_t`；P90/P50/P10固定对应`q0.10/q0.50/q0.90`。界面不得逐体素相加分位数、从边际概率重建空间draw，或依据孔数/颜色/阈值自动显示正式资源量、储量、可采资源量。每个导出物必须带run-id、证据等级、单位、分位点和合规锁状态。
>
> **WP1统计契约**：统一为`calibration_record={calibration_type, nominal_level, sampling_unit, split_id, statistic, estimate, uncertainty_interval, sharpness, status, evidence_id, run_id}`。类型仅允许`ppc`、`holdout_predictive_calibration`、`sbc_rank`、`parameter_interval_coverage`、`field_prediction_interval_coverage`；禁止裸`coverage`。

## 第六节 Python代码示例

### 6.1 数据加载示例

```python
import h5py
import numpy as np
import xarray as xr
from pathlib import Path
from typing import Dict, List, Tuple, Optional
import logging

logger = logging.getLogger(__name__)

class ProbabilityCubeLoader:
    """
    概率立方体数据加载器

    支持HDF5、NetCDF格式，提供统一的xarray接口。
    """

    def __init__(self, filepath: Path):
        """
        初始化加载器

        参数:
            filepath: 数据文件路径
        """
        self.filepath = Path(filepath)
        self._validate_file()

    def _validate_file(self) -> None:
        """验证文件有效性"""
        if not self.filepath.exists():
            raise FileNotFoundError(f"文件不存在: {self.filepath}")

        if self.filepath.suffix not in ['.h5', '.hdf5', '.nc']:
            raise ValueError(f"不支持的文件格式: {self.filepath.suffix}")

    def load_hdf5(self) -> xr.Dataset:
        """
        加载HDF5格式概率立方体

        返回:
            xarray.Dataset对象
        """
        try:
            with h5py.File(self.filepath, 'r') as f:
                # 读取概率数据
                probability = f['probability'][:]
                shape = probability.shape
                nx, ny, nz, n_lith = shape

                # 读取坐标
                x = f['coordinates/x'][:]
                y = f['coordinates/y'][:]
                z = f['coordinates/z'][:]

                # 读取岩性标签
                labels = [l.decode('utf-8') if isinstance(l, bytes) else l
                         for l in f['lithology_labels'][:]]

                # 读取元数据
                attrs = dict(f['metadata'].attrs)

            # 创建xarray Dataset
            ds = xr.Dataset(
                data_vars={
                    'probability': (['x', 'y', 'z', 'lithology'], probability)
                },
                coords={
                    'x': x,
                    'y': y,
                    'z': z,
                    'lithology': labels
                },
                attrs=attrs
            )

            logger.info(f"成功加载概率立方体: shape={shape}")
            return ds

        except Exception as e:
            logger.error(f"加载HDF5文件失败: {e}")
            raise

    def load_netcdf(self) -> xr.Dataset:
        """
        加载NetCDF格式概率立方体

        返回:
            xarray.Dataset对象
        """
        try:
            ds = xr.open_dataset(self.filepath)
            logger.info(f"成功加载NetCDF文件")
            return ds
        except Exception as e:
            logger.error(f"加载NetCDF文件失败: {e}")
            raise

    def load(self) -> xr.Dataset:
        """
        自动识别格式并加载数据

        返回:
            xarray.Dataset对象
        """
        if self.filepath.suffix in ['.h5', '.hdf5']:
            return self.load_hdf5()
        elif self.filepath.suffix == '.nc':
            return self.load_netcdf()
        else:
            raise ValueError(f"不支持的文件格式: {self.filepath.suffix}")

    def get_metadata(self) -> Dict:
        """
        获取元数据信息

        返回:
            元数据字典
        """
        ds = self.load()
        return {
            'shape': ds['probability'].shape,
            'coordinates': {
                'x': (ds.x.min().values, ds.x.max().values),
                'y': (ds.y.min().values, ds.y.max().values),
                'z': (ds.z.min().values, ds.z.max().values),
            },
            'lithologies': ds.coords['lithology'].values.tolist(),
            'resolution': {
                'dx': float(np.diff(ds.x).mean()),
                'dy': float(np.diff(ds.y).mean()),
                'dz': float(np.diff(ds.z).mean()),
            },
            'attrs': ds.attrs
        }


# 使用示例
if __name__ == "__main__":
    loader = ProbabilityCubeLoader(Path("probability_cube.h5"))
    metadata = loader.get_metadata()
    print(f"岩性类型: {metadata['lithologies']}")
    print(f"数据范围: {metadata['coordinates']}")

    ds = loader.load()
    print(f"概率范围: {ds['probability'].min().values:.3f} - {ds['probability'].max().values:.3f}")
```

```python
        初始化渲染器

        参数:
            data: 三维数据数组 (nx, ny, nz) 或 (nx, ny, nz, n_lith)
            spacing: 空间分辨率 (dx, dy, dz)
        """
        self.data = data
        self.spacing = spacing
        self.plotter = None

    def create_grid(self, lithology_idx: Optional[int] = None) -> pv.UniformGrid:
        """
        创建PyVista网格

        参数:
            lithology_idx: 岩性索引（对于多维数据）

        返回:
            UniformGrid对象
        """
        # 处理维度
        if self.data.ndim == 4:
            if lithology_idx is None:
                lithology_idx = 0
            data_3d = self.data[:, :, :, lithology_idx]
        else:
            data_3d = self.data

        # 创建网格
        grid = pv.UniformGrid()
        grid.dimensions = np.array(data_3d.shape) + 1
        grid.spacing = self.spacing
        grid.origin = (0, 0, 0)
        grid.cell_data["values"] = data_3d.flatten(order="F")

        return grid

    def render_volume(
        self,
        lithology_idx: int = 0,
        threshold: Tuple[float, float] = (0.5, 1.0),
        cmap: str = "viridis",
        opacity: str = "linear",
        show: bool = True,
        screenshot: Optional[str] = None
    ) -> Optional[pv.Plotter]:
        """
        体绘制渲染

        参数:
            lithology_idx: 岩性索引
            threshold: 显示阈值范围
            cmap: 颜色映射
            opacity: 不透明度映射方式
            show: 是否显示窗口
            screenshot: 截图保存路径

        返回:
            Plotter对象（当show=False时）
        """
        # 创建网格
        grid = self.create_grid(lithology_idx)

        # 应用阈值
        thresholded = grid.threshold(threshold)

        if thresholded.n_cells == 0:
            logger.warning(f"阈值{threshold}范围内无数据点")
            return None

        # 创建绘图器
        self.plotter = pv.Plotter()
        self.plotter.add_volume(
            thresholded,
            scalars="values",
            cmap=cmap,
            opacity=opacity,
            show_scalar_bar=True
        )

        # 添加坐标轴
        self.plotter.add_axes()

        # 保存截图
        if screenshot is not None:
            self.plotter.screenshot(screenshot)
            logger.info(f"截图已保存: {screenshot}")

        if show:
            self.plotter.show()
            return None
        else:
            return self.plotter

    def render_isosurface(
        self,
        lithology_idx: int = 0,
        value: float = 0.7,
        color: Tuple[float, float, float] = (1.0, 0.5, 0.0),
        opacity: float = 0.8,
        show: bool = True
    ) -> Optional[pv.Plotter]:
        """
        等值面提取与渲染

        参数:
            lithology_idx: 岩性索引
            value: 等值面值
            color: RGB颜色
            opacity: 不透明度
            show: 是否显示窗口

        返回:
            Plotter对象（当show=False时）
        """
        # 创建网格
        grid = self.create_grid(lithology_idx)

        # 提取等值面
        isosurface = grid.contour(isosurfaces=[value])

        if isosurface.n_points == 0:
            logger.warning(f"无法提取值={value}的等值面")
            return None

        # 创建绘图器
        self.plotter = pv.Plotter()
        self.plotter.add_mesh(
            isosurface,
            color=color,
            opacity=opacity,
            show_scalar_bar=False
        )

        # 添加坐标轴
        self.plotter.add_axes()

        # 添加标题
        self.plotter.add_text(
            f"Isosurface (value={value})",
            position='upper_left'
        )

        if show:
            self.plotter.show()
            return None
        else:
            return self.plotter

    def render_slice(
        self,
        lithology_idx: int = 0,
        axis: str = 'z',
        position: Optional[float] = None,
        cmap: str = "RdYlBu_r",
        show: bool = True
    ) -> Optional[pv.Plotter]:
        """
        切片渲染

        参数:
            lithology_idx: 岩性索引
            axis: 切片方向 ('x', 'y', 'z')
            position: 切片位置（None表示中心）
            cmap: 颜色映射
            show: 是否显示窗口

        返回:
            Plotter对象（当show=False时）
        """
        # 创建网格
        grid = self.create_grid(lithology_idx)

        # 确定切片位置
        if position is None:
            center = grid.center
            position = center[['x', 'y', 'z'].index(axis)]

        # 创建切片
        slice_mesh = grid.slice(normal=axis, origin=(0, 0, position))

        # 创建绘图器
        self.plotter = pv.Plotter()
        self.plotter.add_mesh(
            slice_mesh,
            scalars="values",
            cmap=cmap,
            show_scalar_bar=True
        )

        # 添加标题
        self.plotter.add_text(
            f"Slice: {axis}={position:.1f}m",
            position='upper_left'
        )

        if show:
            self.plotter.show()
            return None
        else:
            return self.plotter

    def render_multi_slice(
        self,
        lithology_idx: int = 0,
        axis: str = 'z',
        n_slices: int = 5,
        cmap: str = "RdYlBu_r",
        show: bool = True
    ) -> Optional[pv.Plotter]:
        """
        多切片同时渲染

        参数:
            lithology_idx: 岩性索引
            axis: 切片方向
            n_slices: 切片数量
            cmap: 颜色映射
            show: 是否显示窗口

        返回:
            Plotter对象（当show=False时）
        """
        # 创建网格
        grid = self.create_grid(lithology_idx)

        # 获取范围
        bounds = grid.bounds
        if axis == 'x':
            start, end = bounds[0], bounds[1]
        elif axis == 'y':
            start, end = bounds[2], bounds[3]
        else:  # 'z'
            start, end = bounds[4], bounds[5]

        # 创建等间距切片
        positions = np.linspace(start, end, n_slices)

        # 创建绘图器
        self.plotter = pv.Plotter()

        # 添加所有切片
        for i, pos in enumerate(positions):
            slice_mesh = grid.slice(normal=axis, origin=(0, 0, pos))
            self.plotter.add_mesh(
                slice_mesh,
                scalars="values",
                cmap=cmap,
                show_scalar_bar=(i == 0),  # 只显示第一个的colorbar
                opacity=0.9,
                name=f"slice_{i}"
            )

        if show:
            self.plotter.show()
            return None
        else:
            return self.plotter
```

### 6.3 动画生成示例

```python
import pyvista as pv
import numpy as np
from pathlib import Path
from typing import List

class AnimationGenerator:
    """
    动画生成器

    创建概率立方体的旋转、切片扫描等动画。
    """

    def __init__(self, grid: pv.UniformGrid):
        """
        初始化动画生成器

        参数:
            grid: PyVista网格对象
        """
        self.grid = grid

    def create_rotation_animation(
        self,
        output_path: str,
        frames: int = 60,
        duration: float = 5.0
    ) -> None:
        """
        创建旋转动画

        参数:
            output_path: 输出文件路径(.gif或.mp4)
            frames: 总帧数
            duration: 动画时长(秒)
        """
        plotter = pv.Plotter(off_screen=True)
        plotter.add_volume(self.grid, cmap="hot")

        # 设置相机轨迹
        plotter.open_gif(output_path, fps=frames/duration)
        plotter.write_frame()

        for i in range(frames):
            plotter.camera.azimuth += 360 / frames
            plotter.write_frame()

        plotter.close()
        print(f"旋转动画已保存: {output_path}")

    def create_slice_scan_animation(
        self,
        output_path: str,
        axis: str = 'z',
        frames: int = 60,
        duration: float = 5.0
    ) -> None:
        """
        创建切片扫描动画

        参数:
            output_path: 输出文件路径
            axis: 扫描方向
            frames: 总帧数
            duration: 动画时长(秒)
        """
        plotter = pv.Plotter(off_screen=True)

        # 获取范围
        bounds = self.grid.bounds
        if axis == 'x':
            start, end = bounds[0], bounds[1]
        elif axis == 'y':
            start, end = bounds[2], bounds[3]
        else:
            start, end = bounds[4], bounds[5]

        positions = np.linspace(start, end, frames)

        plotter.open_gif(output_path, fps=frames/duration)

        for pos in positions:
            plotter.clear()
            slice_mesh = self.grid.slice(normal=axis, origin=(0, 0, pos))
            plotter.add_mesh(slice_mesh, scalars="values", cmap="RdYlBu_r")
            plotter.write_frame()

        plotter.close()
        print(f"扫描动画已保存: {output_path}")

    def create_uncertainty_animation(
        self,
        uncertainty_grid: pv.UniformGrid,
        output_path: str,
        frames: int = 60,
        duration: float = 5.0
    ) -> None:
        """
        创建概率-不确定性切换动画

        参数:
            uncertainty_grid: 不确定性网格
            output_path: 输出文件路径
            frames: 总帧数
            duration: 动画时长(秒)
        """
        plotter = pv.Plotter(off_screen=True)
        plotter.open_gif(output_path, fps=frames/duration)

        half_frames = frames // 2

        # 前半段显示概率
        for i in range(half_frames):
            plotter.clear()
            alpha = i / half_frames
            plotter.add_volume(self.grid, cmap="hot", opacity=0.5*alpha)
            plotter.write_frame()

        # 后半段显示不确定性
        for i in range(half_frames):
            plotter.clear()
            alpha = 1 - i / half_frames
            plotter.add_volume(uncertainty_grid, cmap="viridis_r", opacity=0.5*alpha)
            plotter.write_frame()

        plotter.close()
        print(f"切换动画已保存: {output_path}")


# 使用示例
if __name__ == "__main__":
    # 创建测试数据
    nx, ny, nz = 50, 50, 50
    data = np.random.rand(nx, ny, nz)

    grid = pv.UniformGrid()
    grid.dimensions = np.array(data.shape) + 1
    grid.cell_data["values"] = data.flatten(order="F")

    # 生成动画
    animator = AnimationGenerator(grid)
    animator.create_rotation_animation("rotation.gif")
    animator.create_slice_scan_animation("scan.gif", axis='z')
```
